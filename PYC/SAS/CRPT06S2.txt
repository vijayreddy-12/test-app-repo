 %INCLUDE INCLUDE(OPTIONS2);                                                    
 %INCLUDE MACROS(VSAMERRI,FILEREAD);                                            
  /*******************************************************************/         
  /*READ TODAY'S DATE                                                */         
  /*******************************************************************/         
  %FILEREAD(DATES,DATEFILE,DATELOGIC,0);                                        
  DATELOGIC:                                                                    
                                                                                
     %GLOBAL D_TODAY;                                                           
     %GLOBAL D_DATE;                                                            
     %GLOBAL D_CYYMM;                                                           
     %GLOBAL D_MONTH;                                                           
     %GLOBAL D_YEAR;                                                            
     %GLOBAL D_DAY;                                                             
     %GLOBAL MONTH;                                                             
                                                                                
     ATTRIB  MONTH_T   FORMAT=$CHAR10.;                                         
                                                                                
     INPUT @1  COMMENT $1.                                                      
           @2  TYPE    $9.                                                      
           @11 INDATE  $9.                                                      
           @11 INYEAR  $4.                                                      
           @64 INDATE1 $7.                                                      
           @64 INCYYMM $5.                                                      
           @65 INYY1   $2.                                                      
           @67 INMM1   $2.                                                      
           @69 INDD1   $2.                                                      
           @;                                                                   
                                                                                
     IF COMMENT NE '*' AND TYPE EQ 'CURRENT  ' THEN DO;                         
        CALL SYMPUT('D_TODAY',PUT(INDATE1,$7.));                                
        CALL SYMPUT('D_DATE',PUT(INDATE,$9.));                                  
        CALL SYMPUT('D_CYYMM',PUT(INCYYMM,$5.));                                
        CALL SYMPUT('D_YEAR',PUT(INYEAR,$4.));                                  
        CALL SYMPUT('D_DAY',PUT(INDD1,$2.));                                    
     END;                                                                       
     ELSE RETURN;                                                               
                                                                                
     SELECT (INMM1);                                                            
         WHEN ('01') MONTH_T  = 'JANUARY';                                      
         WHEN ('02') MONTH_T  = 'FEBRUARY';                                     
         WHEN ('03') MONTH_T  = 'MARCH';                                        
         WHEN ('04') MONTH_T  = 'APRIL';                                        
         WHEN ('05') MONTH_T  = 'MAY';                                          
         WHEN ('06') MONTH_T  = 'JUNE';                                         
         WHEN ('07') MONTH_T  = 'JULY';                                         
         WHEN ('08') MONTH_T  = 'AUGUST';                                       
         WHEN ('09') MONTH_T  = 'SEPTEMBER';                                    
         WHEN ('10') MONTH_T  = 'OCTOBER';                                      
         WHEN ('11') MONTH_T  = 'NOVEMBER';                                     
         WHEN ('12') MONTH_T  = 'DECEMBER';                                     
         OTHERWISE   MONTH_T  = ' ';                                            
     END;                                                                       
                                                                                
     CALL SYMPUT('MONTH',PUT(MONTH_T,$03.));                                    
                                                                                
     KEEP D_TODAY D_DATE D_CYYMM D_MONTH D_YEAR D_DAY MONTH;                    
    OUTPUT;                                                                     
  RETURN;
