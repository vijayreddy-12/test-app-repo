     PROC SQL;                                                                  
     CREATE TABLE GROUP1  AS                                                    
     SELECT                                                                     
            A.CUST_ID,                                                          
            A.GROUP_ID,                                                         
            A.DIV_ID,                                                           
            A.REST_REC                                                          
       FROM INFIL1 A                                                            
       ;                                                                        
                                                                                
    /* PROC PRINT DATA=PA1; */                                                  
                                                                                
                                                                                
     PROC SQL;                                                                  
     CREATE TABLE PASECUPD1 AS                                                  
     SELECT                                                                     
            B.CUST_ID,                                                          
            B.ACC_IND                                                           
       FROM INFIL2 B                                                            
       ;                                                                        
  /*  PROC PRINT DATA=TLH1;   */                                                
                                                                                
  /*================================================================            
    SQL - PICK UP ONLY THOSE RECORD FROM EMLCNT XTRACR FILE WHERE               
          MATCHING GROUP_ID   EXIST IN BOTH GROUP  EXTRACT AND                  
          EMALCN XTRACT.                                                        
    ===============================================================*/           
     PROC SQL;                                                                  
     CREATE TABLE COMMON1 AS                                                    
     SELECT                                                                     
            A.CUST_ID,                                                          
            A.ACC_IND                                                           
       FROM PASECUPD1 A, GROUP1 B                                               
       WHERE A.CUST_ID  = B.CUST_ID                                             
       ;                                                                        
                                                                                
  PROC SORT DATA = COMMON1 NODUPKEY;                                            
     BY CUST_ID;                                                                
                                                                                
  /* PROC PRINT DATA=COMMON1;  */                                               
                                                                                
  /* ***************************************************** */                   
  /* MERGE MASTER EXTRACT WITH AGENT EXTRACT WHEN THERE IS */                   
  /* MASTCHING AGENT_CODE IN BOTH THE FILE.                */                   
  /* ***************************************************** */                   
                                                                                
  DATA COMMON2;                                                                 
   MERGE GROUP1(IN=K1) COMMON1(IN=K2);                                          
    BY CUST_ID;                                                                 
                                                                                
    IF K1 AND K2 THEN OUTPUT;                                                   
   RETURN;                                                                      
                                                                                
  /* PROC PRINT DATA=COMMON2; */                                                
  /* ***************************************************** */                   
  /* MERGE MASTER EXTRACT WITH AGENT EXTRACT WHEN          */                   
  /* AGENT_CODE EXISTS IN MASTER EXTRACT BUT NOT IN         */                  
  /* AGENT EXTRACT                                          */                  
  /* ***************************************************** */                   
                                                                                
  DATA COMMON3;                                                                 
   MERGE GROUP1(IN=K3) COMMON1(IN=K4);                                          
    BY CUST_ID;                                                                 
                                                                                
    IF K3 AND NOT K4 THEN OUTPUT;                                               
                                                                                
   RETURN;                                                                      
  /* PROC PRINT DATA=COMMON3;  */                                               
                                                                                
  /* ***************************************************** */                   
  /* APPEND AND SORT THE DATA FILE PRODUCE BEFORE          */                   
  /* ALSO, SET PAD STATUS AS 'N' WHEN THE PAD FIELD IS BLANK */                 
  /* ***************************************************** */                   
                                                                                
  DATA COMMON4;                                                                 
  SET COMMON2 COMMON3;                                                          
  /* IF BNK_STATUS EQ " " THEN BNK_STATUS = "N"; */                             
     IF ACC_IND    EQ " " THEN ACC_IND    = "#";                                
  OUTPUT;                                                                       
  RETURN;
