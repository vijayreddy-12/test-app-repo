/*-------------------------------------------------------                       
  - READ TODAY'S DATE                               -                           
  -------------------------------------------------------*/                     
 %FILEREAD(DATES,DATEFILE,DATELOGIC,0);                                         
 DATELOGIC:                                                                     
                                                                                
   %GLOBAL D_START_DATE;                                                        
   %GLOBAL D_END_DATE;                                                          
   ATTRIB START_DATE_MM   FORMAT=$CHAR02.;                                      
   ATTRIB START_DATE_YYYY FORMAT=$CHAR04.;                                      
   ATTRIB START_DATE      FORMAT=$CHAR12.;                                      
   ATTRIB END_DATE        FORMAT=$CHAR12.;                                      
                                                                                
   INPUT @1  COMMENT  $1.                                                       
         @2  TYPE     $9.                                                       
         @38 DATEYYYY $CHAR04.                                                  
         @43 DATEMM   $CHAR02.                                                  
         @46 DATEDD   $CHAR02.                                                  
         @;                                                                     
                                                                                
   IF COMMENT EQ '*' OR TYPE NE 'CURRENT  ' THEN RETURN;                        
                                                                                
   IF DATEMM = '01' THEN START_DATE_MM = '12';                                  
   IF DATEMM = '02' THEN START_DATE_MM = '01';                                  
   IF DATEMM = '03' THEN START_DATE_MM = '02';                                  
   IF DATEMM = '04' THEN START_DATE_MM = '03';                                  
   IF DATEMM = '05' THEN START_DATE_MM = '04';                                  
   IF DATEMM = '06' THEN START_DATE_MM = '05';                                  
   IF DATEMM = '07' THEN START_DATE_MM = '06';                                  
   IF DATEMM = '08' THEN START_DATE_MM = '07';                                  
   IF DATEMM = '09' THEN START_DATE_MM = '08';                                  
   IF DATEMM = '10' THEN START_DATE_MM = '09';                                  
   IF DATEMM = '11' THEN START_DATE_MM = '10';                                  
   IF DATEMM = '12' THEN START_DATE_MM = '11';                                  
                                                                                
   IF START_DATE_MM = '12' THEN                                                 
   DO;                                                                          
      START_DATE_YYYY = PUT(PUT(DATEYYYY,4.) - 1,$04.);                         
   END;                                                                         
   ELSE                                                                         
   DO;                                                                          
      START_DATE_YYYY = DATEYYYY;                                               
   END;                                                                         
                                                                                
   START_DATE = "'" || START_DATE_YYYY|| "-" || START_DATE_MM ||                
                "-01'";                                                         
                                                                                
   END_DATE   = "'" || DATEYYYY || "-" || DATEMM ||                             
                "-25'";                                                         
                                                                                
   CALL SYMPUT('D_START_DATE',PUT(START_DATE,$12.));                            
   CALL SYMPUT('D_END_DATE',PUT(END_DATE,$12.));                                
                                                                                
 /*KEEP D_DATE_TIME; */                                                         
                                                                                
   OUTPUT;                                                                      
                                                                                
 RETURN;

    TC.CUST_ADDR_POST_CD "POSTAL_CD",

    TC.EMAIL_ADDR        "EMAIL ADDRESS",                                       
    TC.BUS_PHONE_NUM     "BUSPHON",                                             
    TC.BUS_PHONE_EXT     "EXTN",                                                
    TC.PHONE_NUM         "PHONE",                                               
    TC.LANG_CD           "LANG"                                                 
    FROM DBCYCP1.TPAA TP                                                        
    Join DBCYCP1.TCUST TC                                                       
      On TP.CUST_ID  = TC.CUST_ID                                               
    Join DBCYCP1.TGD    TG                                                      
      On TP.GROUP_ID = TG.GROUP_ID                                              
      AND TP.DIV_ID  = TG.DIV_ID                                                
    Join DBCYCP1.TUT    TU                                                      
      On TU.CUST_ID  = TC.CUST_ID                                               
    Join DBCYCP1.TEH    TEH                                                     
      On TC.CUST_ID  = TEH.CUST_ID                                              
    WHERE TC.ROLE             = 'P'                                             
      AND TP.PA_STAT_CD       = 'A'                                             
      AND TG.BUS_SEG_CD       = 'A'                                             
      AND TG.CONTRACT_STAT_CD = 'I'                                             
      AND (TEH.EVENT_TYP_CD   = 'PA'                                            
           AND (DATE(TEH.EVENT_TS) BETWEEN &D_START_DATE                        
                AND &D_END_DATE                                                 
               )                                                                
          )                                                                     
    ORDER BY TP.GROUP_ID,                                                       
             TP.DIV_ID                                                          
   );
