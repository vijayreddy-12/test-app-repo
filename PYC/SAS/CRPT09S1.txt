  /*-------------------------------------------------------             00860000
    - GET ADVISOR DETAILS                                 -             00870000
    -------------------------------------------------------*/           00880000
                                                                                
  OPTIONS LINESIZE=100 DB2SSID=PDBC NOCENTER;                                   
                                                                         0085000
  PROC SQL;                                                                     
     CREATE TABLE ADVQUERY AS                                                   
     SELECT *                                                                   
     FROM CONNECTION TO DB2                                                     
     (SELECT TC.CUST_ID         AS ADVID                                        
            ,TC.CUST_FIRST_NAME AS ADVFNM                                       
            ,TC.CUST_LAST_NAME  AS ADVLNM                                       
            ,TA.ADVISOR_ID      AS SELLCD                                       
            ,YEAR(TA.CHN_TS)    AS SCYYYY                                       
            ,MONTH(TA.CHN_TS)   AS SCMM                                         
            ,DAY(TA.CHN_TS)     AS SCDD                                         
            ,HOUR(TA.CHN_TS)    AS SCHH                                         
            ,MINUTE(TA.CHN_TS)  AS SCMN                                         
            ,TC.CHN_USER_ID     AS CHUID                                        
            ,YEAR(TC.CHN_TS)    AS CHYYYY                                       
            ,MONTH(TC.CHN_TS)   AS CHMM                                         
            ,DAY(TC.CHN_TS)     AS CHDD                                         
            ,HOUR(TC.CHN_TS)    AS CHHH                                         
            ,MINUTE(TC.CHN_TS)  AS CHMN                                         
      FROM DBCYCP1.TCUST TC JOIN DBCYCP1.TADVSR TA                              
        ON TC.CUST_ID = TA.CUST_ID                                              
      WHERE TA.CHN_TS > &D_DATE_TIME                                            
     );                                                                         
                                                                                
  PROC PRINT DATA=ADVQUERY;                                                     
                                                                                
  /*-------------------------------------------------------             00860000
    - FORMAT CHANGE TIMESTAMPS                            -             00870000
    -------------------------------------------------------*/           00880000
                                                                                
  DATA ADVLIST;                                                                 
    SET ADVQUERY;                                                               
                                                                                
     ATTRIB SCDYYYY FORMAT=$CHAR4.                                              
            SCDMM   FORMAT=$CHAR2.                                              
            SCDDD   FORMAT=$CHAR2.                                              
            SCDHH   FORMAT=$CHAR2.                                              
            SCDMN   FORMAT=$CHAR2.                                              
            SCTS    FORMAT=$CHAR25.;                                            
                                                                                
     SCDYYYY = PUT(SCYYYY, $04.);                                               
                                                                                
     IF SCMM < 10 THEN                                                          
     DO;                                                                        
        SCDMM = '0' || PUT(SCMM, $01.);                                         
     END;                                                                       
     ELSE                                                                       
     DO;                                                                        
        SCDMM = PUT(SCMM, $02.);                                                
     END;                                                                       
                                                                                
     IF SCDD < 10 THEN                                                          
     DO;                                                                        
        SCDDD   = '0'|| PUT(SCDD, $01.);                                        
     END;                                                                       
     ELSE                                                                       
     DO;                                                                        
        SCDDD   = PUT(SCDD, $02.);                                              
     END;                                                                       
                                                                                
     IF SCHH < 10 THEN                                                          
     DO;                                                                        
        SCDHH   = '0' || PUT(SCHH, $01.);                                       
     END;                                                                       
     ELSE                                                                       
     DO;                                                                        
        SCDHH   = PUT(SCHH, $02.);                                              
     END;                                                                       
                                                                                
     IF SCMN < 10 THEN                                                          
     DO;                                                                        
        SCDMN   = '0' || PUT(SCMN, $01.);                                       
     END;                                                                       
     ELSE                                                                       
     DO;                                                                        
        SCDMN   = PUT(SCMN, $02.);                                              
     END;                                                                       
                                                                                
     SCTS = SCDYYYY || '-' || SCDMM || '-' || SCDDD || ' ' ||                   
            SCDHH || ':' || SCDMN;                                              
                                                                                
     ATTRIB CHDYYYY FORMAT=$CHAR4.                                              
            CHDMM   FORMAT=$CHAR2.                                              
            CHDDD   FORMAT=$CHAR2.                                              
            CHDHH   FORMAT=$CHAR2.                                              
            CHDMN   FORMAT=$CHAR2.                                              
            CHTS    FORMAT=$CHAR16.;                                            
                                                                                
     CHDYYYY = PUT(CHYYYY, $04.);                                               
                                                                                
     IF CHMM < 10 THEN                                                          
     DO;                                                                        
        CHDMM = '0' || PUT(CHMM, $01.);                                         
     END;                                                                       
     ELSE                                                                       
     DO;                                                                        
        CHDMM = PUT(CHMM, $02.);                                                
     END;                                                                       
                                                                                
     IF CHDD < 10 THEN                                                          
     DO;                                                                        
        CHDDD   = '0'|| PUT(CHDD, $01.);                                        
     END;                                                                       
     ELSE                                                                       
     DO;                                                                        
        CHDDD   = PUT(CHDD, $02.);                                              
     END;                                                                       
                                                                                
     IF CHHH < 10 THEN                                                          
     DO;                                                                        
        CHDHH   = '0' || PUT(CHHH, $01.);                                       
     END;                                                                       
     ELSE                                                                       
     DO;                                                                        
        CHDHH   = PUT(CHHH, $02.);                                              
     END;                                                                       
                                                                                
     IF CHMN < 10 THEN                                                          
     DO;                                                                        
        CHDMN   = '0' || PUT(CHMN, $01.);                                       
     END;                                                                       
     ELSE                                                                       
     DO;                                                                        
        CHDMN   = PUT(CHMN, $02.);                                              
     END;                                                                       
                                                                                
     CHTS = CHDYYYY || '-' || CHDMM || '-' || CHDDD || ' ' ||                   
            CHDHH || ':' || CHDMN;                                              
                                                                                
    OUTPUT;                                                                     
    RETURN;
