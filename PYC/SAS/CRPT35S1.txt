  %GLOBAL D_START_DATE;                                                         
  %GLOBAL D_END_DATE;                                                           
  ATTRIB START_DATE      FORMAT=$CHAR12.;                                       
  ATTRIB END_DATE        FORMAT=$CHAR12.;                                       
                                                                                
  INPUT @1  START_DATE   $CHAR12.                                               
        @15 END_DATE     $CHAR12.                                               
        @;                                                                      
                                                                                
   PUT " ";

   PUT "  END DATE IS " END_DATE;                                               
   PUT " ";                                                                     
                                                                                
  CALL SYMPUT('D_START_DATE',PUT(START_DATE,$12.));                             
  CALL SYMPUT('D_END_DATE',PUT(END_DATE,$12.));                                 
                                                                                
PROC SQL;                                                                       
   CREATE TABLE GRPLIST AS                                                      
   SELECT *                                                                     
   FROM CONNECTION TO DB2                                                       
   (SELECT DISTINCT                                                             
    TU.CUST_ID "CUST-ID",                                                       
    TU.USER_ID "USER",                                                          
    TU.MAPPED_ID "LINK_REPSRC_ID",                                              
    TP.GROUP_ID "GROUP",                                                        
    TP.DIV_ID "DIVISION",                                                       
    TG.BUS_SEG_CD "BUSSEG",                                                     
    TG.ACC_PM_REGIS_CD "ACCESS_CD",                                             
    TC.CUST_FIRST_NAME "FIRST_NAME",                                            
    TC.CUST_LAST_NAME "LAST_NAME",                                              
    TG.REG_GROUP_OFFICE "RGO",                                                  
    TC.ROLE "ROLE",                                                             
    TC.CUST_STAT_CD "STATUS",                                                   
    TC.CUST_STAT_REAS_CD "REASON",                                              
    TG.SPONSOR_NAME "SPONSOR",                                                  
    TC.CUST_ADDR_1 "ADDR_1",                                                    
    TC.CUST_ADDR_2 "ADDR_2",                                                    
    TC.CUST_ADDR_CITY "CITY",                                                   
    TC.CUST_ADDR_PROV "PROVINCE",                                               
    TC.CUST_ADDR_COUNTRY "COUNTRY",                                             
    TC.CUST_ADDR_POST_CD "POSTAL_CD",                                           
    DATE(TC.CHN_TS) "UPDDAT1",                                                  
    TC.EMAIL_ADDR "EMAIL ADDRESS"                                               
    FROM DBCYCP1.TUT TU,                                                        
         DBCYCP1.TCUST TC,                                                      
         DBCYCP1.TPAA TP,                                                       
         DBCYCP1.TGD  TG                                                        
    WHERE                                                                       
          TU.CUST_ID = TC.CUST_ID AND                                           
          TC.CUST_ID = TP.CUST_ID AND                                           
          TC.ROLE    = 'P' AND                                                  
          TC.CUST_STAT_REAS_CD = 'A' AND                                        
          TP.GROUP_ID = TG.GROUP_ID AND                                         
          TP.DIV_ID   = TG.DIV_ID AND                                           
           (DATE(TC.CHN_TS) > &D_START_DATE AND                                 
           DATE(TC.CHN_TS) < &D_END_DATE)                                       
    ORDER BY TG.BUS_SEG_CD,                                                     
             TG.REG_GROUP_OFFICE,                                               
             TP.GROUP_ID,                                                       
             TP.DIV_ID,                                                         
             TU.USER_ID                                                         
   );
