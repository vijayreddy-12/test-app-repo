OPTIONS LINESIZE=120 DB2SSID=PDBC NOCENTER;                                     
                                                                                
PROC SQL;                                                                       
   CREATE TABLE GRPLIST AS                                                      
   SELECT *                                                                     
   FROM CONNECTION TO DB2                                                       
   (SELECT DISTINCT                                                             
    TU.USER_ID "USER",                                                          
    TP.GROUP_ID "GROUP",                                                        
    TP.DIV_ID "DIVISION",                                                       
    TC.CUST_FIRST_NAME "FIRST_NAME",                                            
    TC.CUST_LAST_NAME "LAST_NAME",                                              
    TC.ROLE "ROLE",                                                             
    TC.CUST_STAT_CD "STATUS",                                                   
    TC.LANG_CD           "LANGCD",                                              
    TC.CUST_ADDR_1 "ADDR_1",                                                    
    TC.CUST_ADDR_2 "ADDR_2",                                                    
    TC.CUST_ADDR_CITY "CITY",                                                   
    TC.CUST_ADDR_PROV "PROVINCE",                                               
    TC.CUST_ADDR_COUNTRY "COUNTRY",                                             
    TC.CUST_ADDR_POST_CD "POSTAL_CD",                                           
    TC.EMAIL_ADDR "EMAIL ADDRESS"                                               
    FROM DBCYCP1.TUT TU,                                                        
         DBCYCP1.TCUST TC,                                                      
         DBCYCP1.TPAA TP,                                                       
         DBCYCP1.TGD  TG                                                        
    WHERE                                                                       
          TU.CUST_ID = TC.CUST_ID AND                                           
          TC.CUST_ID = TP.CUST_ID AND                                           
          TC.ROLE    = 'P' AND                                                  
          TP.GROUP_ID = TG.GROUP_ID AND                                         
          TP.DIV_ID   = TG.DIV_ID                                               
    ORDER BY TU.USER_ID                                                         
   );                                                                           
                                                                                
     ATTRIB ROLE_EXP FORMAT=$30.;                                               
     ATTRIB STATUS_EXP  FORMAT=$10.;                                            
                                                                                
 DATA TEMP1;                                                                    
   SET GRPLIST;                                                                 
  IF STATUS = "A" THEN STATUS_EXP = "Active    ";                               
  IF STATUS = "S" THEN STATUS_EXP = "Suspended ";                               
  IF ROLE = " " THEN ROLE_EXP = "Default value for TGCT        ";               
  IF ROLE = "A" THEN ROLE_EXP = "Plan Advisor                  ";               
  IF ROLE = "C" THEN ROLE_EXP = "CSR                           ";               
  IF ROLE = "I" THEN ROLE_EXP = "Internal Manulife Staff       ";               
  IF ROLE = "M" THEN ROLE_EXP = "Plan Advisor Manager          ";               
  IF ROLE = "P" THEN ROLE_EXP = "Plan Administrator            ";               
  IF ROLE = "R" THEN ROLE_EXP = "RGO Staff                     ";               
  IF ROLE = "S" THEN ROLE_EXP = "CSR with Access to Staff Plan ";               
  IF ROLE = "T" THEN ROLE_EXP = "MLI (Trainer)                 ";               
   OUTPUT;                                                                      
   RETURN;                                                                      
                                                                                
 DATA REPORT;                                                                   
   FILE REPORT;                                                                 
   SET TEMP1;                                                                   
    IF _N_=1 THEN DO;                                                           
                                                                                
    PUT 'USER_ID~First_Name~Last_Name~ROLE~Status~Lang~GRP#~DIV~'               
        'ADDR-1~ADDR-2~City~PROV~Country~Postal_Code~E-mail';                   
        END;                                                                    
                                                                                
   PUT   @1    USER                       20.                                   
         @21   '~'                                                              
         @22   FIRST_NAME                 30.                                   
         @52   '~'                                                              
         @53   LAST_NAME                  30.                                   
         @83   '~'                                                              
         @84   ROLE_EXP                   30.                                   
         @114  '~'                                                              
         @115  STATUS_EXP                 10.                                   
         @125  '~'                                                              
         @126  LANGCD                     01.                                   
         @127  '~'                                                              
         @128  GROUP                      07.                                   
         @135  '~'                                                              
         @136  DIVISION                   03.                                   
         @139  '~'                                                              
         @140  ADDR_1                     60.                                   
         @200  '~'                                                              
         @201  ADDR_2                     60.                                   
         @261  '~'                                                              
         @262  CITY                       60.                                   
         @322  '~'                                                              
         @323  PROVINCE                   30.                                   
         @353  '~'                                                              
         @354  COUNTRY                    30.                                   
         @384  '~'                                                              
         @385  POSTAL_CD                  09.                                   
         @394  '~'                                                              
         @395  EMAIL_ADDRESS              60.                                   
         @455  '~'                                                              
          ;                                                                     
      RUN;
